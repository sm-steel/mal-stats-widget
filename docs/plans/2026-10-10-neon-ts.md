```ts
/** "today", "5h ago", "yesterday", "3d ago", "2w ago", "1mo ago", "1y ago" (as the Python widget did). */
export function relativeTime(
	iso: string | undefined,
	now: Date = new Date(),
): string {
	const t = iso ? Date.parse(iso) : Number.NaN;
	if (Number.isNaN(t)) return "";
	const seconds = (now.getTime() - t) / 1000;
	const days = Math.floor(seconds / 86400);
	if (days < 1) {
		const hours = Math.floor(seconds / 3600);
		return hours < 1 ? "today" : `${hours}h ago`;
	}
	if (days === 1) return "yesterday";
	if (days < 7) return `${days}d ago`;
	if (days < 30) return `${Math.floor(days / 7)}w ago`;
	if (days < 365) return `${Math.floor(days / 30)}mo ago`;
	return `${Math.floor(days / 365)}y ago`;
}
```

```ts
// MyAnimeList data for the stats card: the user's whole anime list via the official
// v2 API (Client-ID auth only), stats computed from it, and poster thumbnails.

export interface Entry {
	node: {
		title?: string;
		main_picture?: { medium?: string };
		average_episode_duration?: number;
	};
	list_status?: {
		status?: string;
		score?: number;
		num_episodes_watched?: number;
		updated_at?: string;
	};
}

export type Reason =
	| "no client id"
	| "no username"
	| "timeout"
	| "bad response"
	| `mal ${number}`;

export class MalError extends Error {
	constructor(readonly reason: Reason) {
		super(reason);
		this.name = "MalError";
	}
}

const API = "https://api.myanimelist.net/v2/";
const CDN = "https://cdn.myanimelist.net/";

const isTimeout = (e: unknown) =>
	e instanceof Error && (e.name === "TimeoutError" || e.name === "AbortError");

export async function fetchAnimeList(
	username: string | undefined,
	clientId: string | undefined,
	fetchFn: typeof fetch = fetch,
	opts: { pageTimeoutMs?: number; totalTimeoutMs?: number; maxPages?: number } = {},
): Promise<Entry[]> {
	const { pageTimeoutMs = 6000, totalTimeoutMs = 18000, maxPages = 10 } = opts;
	if (!clientId) throw new MalError("no client id");
	if (!username) throw new MalError("no username");
	const overall = AbortSignal.timeout(totalTimeoutMs);
	const entries: Entry[] = [];
	let url: string | undefined =
		`${API}users/${encodeURIComponent(username)}/animelist?fields=list_status,average_episode_duration&limit=1000`;
	for (let page = 0; url && page < maxPages; page++) {
		if (!url.startsWith(API)) throw new MalError("bad response");
		let body: { data?: unknown; paging?: { next?: unknown } };
		try {
			const res = await fetchFn(url, {
				headers: { "X-MAL-Client-ID": clientId },
				signal: AbortSignal.any([overall, AbortSignal.timeout(pageTimeoutMs)]),
			});
			if (!res.ok) throw new MalError(`mal ${res.status}`);
			body = await res.json();
		} catch (e) {
			if (e instanceof MalError) throw e;
			throw new MalError(isTimeout(e) ? "timeout" : "bad response");
		}
		if (!Array.isArray(body?.data)) throw new MalError("bad response");
		entries.push(...(body.data as Entry[]));
		const next = body.paging?.next;
		url = typeof next === "string" ? next : undefined;
	}
	return entries;
}

export interface Stats {
	total: number;
	completed: number;
	watching: number;
	meanScore: number;
	daysWatched: number;
}

export function computeStats(entries: Entry[]): Stats {
	const count = (s: string) => entries.filter((e) => e.list_status?.status === s).length;
	const scores = entries.map((e) => e.list_status?.score ?? 0).filter((s) => s > 0);
	const seconds = entries.reduce(
		(sum, e) => sum + (e.list_status?.num_episodes_watched ?? 0) * (e.node.average_episode_duration ?? 0),
		0,
	);
	return {
		total: entries.length,
		completed: count("completed"),
		watching: count("watching"),
		meanScore: scores.length ? scores.reduce((a, b) => a + b, 0) / scores.length : 0,
		daysWatched: seconds / 86400,
	};
}

/** Most recently updated entries, excluding plan-to-watch. */
export function recentEntries(entries: Entry[], count = 6): Entry[] {
	return entries
		.filter((e) => e.list_status?.status !== "plan_to_watch")
		.sort((a, b) => (b.list_status?.updated_at ?? "").localeCompare(a.list_status?.updated_at ?? ""))
		.slice(0, count);
}

/** The CDN's 116×180 resize of a poster (≈7.6 KB instead of ≈57 KB); other hosts unchanged. */
export function posterUrl(medium: string | undefined): string | undefined {
	if (!medium) return undefined;
	return medium.startsWith(CDN) && !medium.startsWith(`${CDN}r/`)
		? `${CDN}r/116x180/${medium.slice(CDN.length)}`
		: medium;
}

const IMAGE_TYPES: Record<string, string> = {
	"image/jpeg": "image/jpeg",
	"image/jpg": "image/jpeg",
	"image/png": "image/png",
	"image/webp": "image/webp",
};

/** Each entry's poster as a data URI, or undefined when missing or the download fails. */
export function fetchPosters(
	entries: Entry[],
	fetchFn: typeof fetch = fetch,
	timeoutMs = 4000,
): Promise<(string | undefined)[]> {
	return Promise.all(
		entries.map(async (e) => {
			const url = posterUrl(e.node.main_picture?.medium);
			if (!url) return undefined;
			try {
				const res = await fetchFn(url, { signal: AbortSignal.timeout(timeoutMs) });
				const type = IMAGE_TYPES[(res.headers.get("content-type") ?? "").split(";")[0]?.trim().toLowerCase() ?? ""];
				if (!res.ok || !type) return undefined;
				return `data:${type};base64,${Buffer.from(await res.arrayBuffer()).toString("base64")}`;
			} catch {
				return undefined;
			}
		}),
	);
}
```

```ts
import {
	type CardOptions,
	card,
	charsPerLine,
	esc,
	FONT_STACK,
	type Palette,
	r3,
	wrapLines,
} from "@sm-steel/neon-kit";
import type { Entry, Stats } from "./mal.js";
import { relativeTime } from "./time.js";

export const LAYOUT = {
	width: 868,
	tileGap: 10,
	tileH: 64,
	subY: 94,
	gridTop: 108,
	colGap: 24,
	rowH: 84,
	rowGap: 12,
	posterW: 60,
	posterH: 84,
	textDx: 74,
	textW: 348,
} as const;

const FONT: CardOptions["font"] = { variant: "text", weights: [400, 700] };
const TITLE_CHARS = charsPerLine(LAYOUT.textW, 13);
const COL_W = (LAYOUT.width - LAYOUT.colGap) / 2;
const TILE_W = (LAYOUT.width - 4 * LAYOUT.tileGap) / 5;

function tiles(s: Stats, p: Palette): string {
	const values: [string, string][] = [
		[String(s.total), "Total"],
		[String(s.completed), "Completed"],
		[String(s.watching), "Watching"],
		[s.meanScore.toFixed(1), "Mean Score"],
		[s.daysWatched.toFixed(1), "Days Watched"],
	];
	return values
		.map(([v, label], i) => {
			const x = r3(i * (TILE_W + LAYOUT.tileGap));
			const cx = r3(x + TILE_W / 2);
			return (
				`<rect x="${x}" y="0" width="${r3(TILE_W)}" height="${LAYOUT.tileH}" rx="2" fill="none" stroke="${p.border}"/>` +
				`<text x="${cx}" y="32" text-anchor="middle" font-size="26" font-weight="700" fill="${p.accent}">${v}</text>` +
				`<text x="${cx}" y="52" text-anchor="middle" font-size="13" fill="${p.text}">${label}</text>`
			);
		})
		.join("");
}

function poster(i: number, href: string | undefined, x: number, y: number, p: Palette): string {
	const box = `x="${x}" y="${y}" width="${LAYOUT.posterW}" height="${LAYOUT.posterH}"`;
	if (!href) return `<rect ${box} rx="2" fill="${p.border}"/>`;
	return (
		`<clipPath id="mal-p${i}"><rect ${box} rx="2"/></clipPath>` +
		`<image ${box} href="${href}" clip-path="url(#mal-p${i})" preserveAspectRatio="xMidYMid slice"/>` +
		`<rect ${box} rx="2" fill="none" stroke="${p.border}"/>`
	);
}

function grid(recent: Entry[], posters: (string | undefined)[], p: Palette, now: Date): string {
	const header =
		`<text x="0" y="${LAYOUT.subY}" font-size="10" letter-spacing="1.2" fill="${p.text}">` +
		`<tspan fill="${p.accent}">//</tspan> RECENTLY WATCHED</text>`;
	const cells = recent
		.map((e, i) => {
			const x = r3((i % 2) * (COL_W + LAYOUT.colGap));
			const y = LAYOUT.gridTop + Math.floor(i / 2) * (LAYOUT.rowH + LAYOUT.rowGap);
			const tx = r3(x + LAYOUT.textDx);
			const lines = wrapLines(e.node.title ?? "", TITLE_CHARS, 3);
			const title = lines
				.map((l, k) => `<text x="${tx}" y="${y + 13 + k * 16}" font-size="13" fill="${p.text}">${esc(l)}</text>`)
				.join("");
			const date = `<text x="${tx}" y="${y + 13 + lines.length * 16}" font-size="11" fill="${p.subtext}">${esc(relativeTime(e.list_status?.updated_at, now))}</text>`;
			return poster(i, posters[i], x, y, p) + title + date;
		})
		.join("");
	return header + cells;
}

export function renderStats(
	stats: Stats,
	recent: Entry[],
	posters: (string | undefined)[],
	p: Palette,
	now: Date = new Date(),
): string {
	const rows = Math.ceil(recent.length / 2);
	const height = rows
		? LAYOUT.gridTop + rows * LAYOUT.rowH + (rows - 1) * LAYOUT.rowGap
		: LAYOUT.tileH;
	return card(
		{ width: LAYOUT.width, height, palette: p, seed: "mal", label: "MyAnimeList anime stats", title: "Anime Stats", font: FONT },
		`<g font-family="${FONT_STACK}">${tiles(stats, p)}${rows ? grid(recent, posters, p, now) : ""}</g>`,
	);
}

export function renderFallback(reason: string, p: Palette): string {
	return card(
		{ width: LAYOUT.width, height: 20, palette: p, seed: "mal", label: `MyAnimeList anime stats: unavailable (${reason})`, title: "Anime Stats", font: FONT },
		`<text x="0" y="15" font-family="${FONT_STACK}" font-size="13" fill="${p.subtext}"><tspan fill="${p.accent}">// SIGNAL LOST</tspan> · ${esc(reason)}</text>`,
	);
}
```

```ts
// GET /api/mal?theme=dark|light — MyAnimeList anime stats as a neon card.
import { FLAVORS } from "@sm-steel/neon-kit";
import { computeStats, fetchAnimeList, fetchPosters, MalError, recentEntries } from "../src/mal.js";
import { renderFallback, renderStats } from "../src/render.js";

export const CACHE_OK = "public, max-age=0, s-maxage=1800, stale-while-revalidate=3600";
export const CACHE_FAIL = "no-store";

export interface Deps {
	fetchFn: typeof fetch;
	env: Record<string, string | undefined>;
	now: () => Date;
}

const svg = (body: string, cache: string) =>
	new Response(body, {
		status: 200,
		headers: { "Content-Type": "image/svg+xml; charset=utf-8", "Cache-Control": cache },
	});

export async function handle(request: Request, deps: Partial<Deps> = {}): Promise<Response> {
	const { fetchFn = fetch, env = process.env, now = () => new Date() } = deps;
	const p = new URL(request.url).searchParams.get("theme") === "light" ? FLAVORS.latte : FLAVORS.mocha;
	try {
		const entries = await fetchAnimeList(env.MAL_USERNAME, env.MAL_CLIENT_ID, fetchFn);
		const recent = recentEntries(entries);
		const posters = await fetchPosters(recent, fetchFn);
		return svg(renderStats(computeStats(entries), recent, posters, p, now()), CACHE_OK);
	} catch (e) {
		// only our own reason strings reach the card, never raw error text
		const reason = e instanceof MalError ? e.reason : "bad response";
		return svg(renderFallback(reason, p), CACHE_FAIL);
	}
}

export function GET(request: Request): Promise<Response> {
	return handle(request);
}
```
